# Create it once (Python 3.12 to match the system ROS Jazzy interpreter):
#

#   source .yolo_moveit/bin/activate
#   pip install -r requirements.txt

# giving us a private ultralytics/torch. NumPy is pinned to the 1.x ABI because
# cv_bridge / system extensions are built against NumPy 1.x — do not bump to 2.x.
